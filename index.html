// ═══════════ TYPOWE DZIADY — AUTOMATYCZNE KURSY (Supabase Edge Function „sync-odds”) ═══════════ // Pobiera kursy 1X2 z The Odds API (bukmacherzy z regionu EU, mediana) i aktualizuje nadchodzące mecze. // Uruchamiana przez harmonogram w bazie (nagłówek x-cron-secret) albo przyciskiem admina w aplikacji. // Zasady (pilnuje też baza): mecze rozpoczęte i z ręcznymi kursami nie są zmieniane. import { createClient } from "npm:@supabase/supabase-js@2"; export const SPORT_KEYS: Record = { "Ekstraklasa": "soccer_poland_ekstraklasa", "Premier League": "soccer_epl", "La Liga": "soccer_spain_la_liga", "Serie A": "soccer_italy_serie_a", "Liga Mistrzów": "soccer_uefa_champs_league", }; const DAYS_AHEAD = 14; // pobieramy tylko ligi z meczami w najbliższych dniach (oszczędza limit API) const ADMIN_MIN_GAP_MIN = 30; // przycisk admina: nie częściej niż co 30 minut // ── Dopasowanie nazw drużyn ─────────────────────────────────────────────────── const DROP = new Set(["fc", "cf", "ac", "as", "ssc", "afc", "sc", "ss", "us", "cd", "ud", "rcd", "sd", "ca", "club", "calcio", "1907", "1913", "1919"]); const WORDS: Record = { warsaw: "warszawa", munich: "monachium", munchen: "monachium", krakow: "krakow", cracow: "krakow", internazionale: "inter" }; // Pełne nazwy po normalizacji, które trzeba zamienić (np. „inter milan” ≠ „milan”) const NAMES: Record = { "inter milan": "inter", "inter": "inter", "milan": "milan", "termalica": "bruk bet termalica nieciecza", "bruk bet termalica": "bruk bet termalica nieciecza", "nieciecza": "bruk bet termalica nieciecza", "psg": "paris saint germain", "man united": "manchester united", "man city": "manchester city", "spurs": "tottenham hotspur", "wolves": "wolverhampton wanderers", "nottm forest": "nottingham forest", "alaves": "deportivo alaves", "deportivo": "deportivo la coruna", "athletic club": "athletic bilbao", "atletico": "atletico madrid", "roma": "roma", "hellas verona": "verona" }; export function normalizeTeam(name: string): string { const base = (name || "").toLowerCase() .replace(/ł/g, "l").replace(/ø/g, "o").replace(/ß/g, "ss") .normalize("NFD").replace(/[̀-ͯ]/g, "") .replace(/&/g, " and ").replace(/[^a-z0-9 ]+/g, " "); const tokens = base.split(/\s+/).filter(t => t && !DROP.has(t)).map(t => WORDS[t] || t); const joined = tokens.join(" "); return NAMES[joined] || joined; } // Ta sama drużyna: identyczna nazwa po normalizacji albo wszystkie słowa krótszej są w dłuższej export function sameTeam(a: string, b: string): boolean { const x = normalizeTeam(a), y = normalizeTeam(b); if (!x || !y) return false; if (x === y) return true; const xs = x.split(" "), ys = y.split(" "); const [short, long] = xs.length <= ys.length ? [xs, ys] : [ys, xs]; return short.every(t => long.includes(t)) && short.join("").length >= 4; } // ── Czas Warszawy ───────────────────────────────────────────────────────────── const warsawParts = (d: Date) => new Intl.DateTimeFormat("sv-SE", { timeZone: "Europe/Warsaw", year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", }).format(d).replace(" ", "T"); export const warsawDate = (iso: string) => warsawParts(new Date(iso)).slice(0, 10); const addDays = (iso: string, n: number) => { const d = new Date(`${iso}T12:00:00Z`); d.setUTCDate(d.getUTCDate() + n); return d.toISOString().slice(0, 10); }; const dayDiff = (a: string, b: string) => Math.round((Date.parse(`${a}T12:00:00Z`) - Date.parse(`${b}T12:00:00Z`)) / 86400000); // ── Kursy z odpowiedzi API: mediana bukmacherów ─────────────────────────────── const median = (xs: number[]) => { const s = [...xs].sort((a, b) => a - b), m = Math.floor(s.length / 2); return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; }; const round2 = (x: number) => Math.round(x * 100) / 100; export function eventOdds(ev: any) { const h: number[] = [], d: number[] = [], a: number[] = []; for (const bm of ev.bookmakers || []) { const mk = (bm.markets || []).find((m: any) => m.key === "h2h"); if (!mk) continue; const price = (n: string) => mk.outcomes.find((o: any) => o.name === n)?.price; const ph = price(ev.home_team), pd = price("Draw"), pa = price(ev.away_team); if (ph > 1 && pd > 1 && pa > 1) { h.push(ph); d.push(pd); a.push(pa); } } if (!h.length) return null; return { odds_home: round2(median(h)), odds_draw: round2(median(d)), odds_away: round2(median(a)), bookmakers: h.length }; } // Mecz z aplikacji → wydarzenie z API (ta sama liga, data ±1 dzień, obie drużyny) export function findEvent(match: any, events: any[]) { if (match.odds_api_id) { const byId = events.find(e => e.id === match.odds_api_id); if (byId) return { event: byId }; } let best: any = null, swapped: any = null; for (const ev of events) { const diff = Math.abs(dayDiff(warsawDate(ev.commence_time), match.match_date)); if (diff > 1) continue; if (sameTeam(match.home, ev.home_team) && sameTeam(match.away, ev.away_team)) { if (!best || diff < best.diff) best = { event: ev, diff }; } else if (sameTeam(match.home, ev.away_team) && sameTeam(match.away, ev.home_team)) { swapped = ev; } } return best ? { event: best.event } : { swapped }; } // ── Główna logika ───────────────────────────────────────────────────────────── // store: { leagues(), upcoming(from, to), update(id, patch), log(row) } export async function syncOdds({ store, apiKey, fetchFn = fetch, now = new Date(), trigger = "cron" }: any) { const nowKey = warsawParts(now), today = nowKey.slice(0, 10); const leagues = await store.leagues(); const matches = await store.upcoming(today, addDays(today, DAYS_AHEAD)); const result = { trigger, updated: 0, matched: 0, unmatched: [] as string[], credits_remaining: null as number | null, errors: [] as string[], leagues: [] as string[] }; for (const lg of leagues) { const key = SPORT_KEYS[lg.name]; const todo = matches.filter((m: any) => m.league_id === lg.id && !m.odds_manual && `${m.match_date}T${m.match_time?.slice(0, 5)}` > nowKey); if (!key || !todo.length) continue; result.leagues.push(lg.name); let events: any[] = []; try { const url = `https://api.the-odds-api.com/v4/sports/${key}/odds?apiKey=${encodeURIComponent(apiKey)}®ions=eu&markets=h2h&oddsFormat=decimal&dateFormat=iso`; const res = await fetchFn(url); const left = res.headers?.get?.("x-requests-remaining"); if (left != null && left !== "") result.credits_remaining = Math.floor(Number(left)); if (!res.ok) { const msg = res.status === 401 ? "nieprawidłowy klucz API" : res.status === 429 ? "wyczerpany limit zapytań" : `błąd ${res.status}`; result.errors.push(`${lg.name}: ${msg}`); continue; } events = await res.json(); } catch (e) { result.errors.push(`${lg.name}: brak połączenia z API (${(e as Error).message})`); continue; } for (const m of todo) { const label = `${lg.name}: ${m.home} – ${m.away} (${m.match_date.slice(8, 10)}.${m.match_date.slice(5, 7)})`; const { event, swapped } = findEvent(m, events) as any; if (!event) { result.unmatched.push(swapped ? `${label} — w API gospodarz i gość są zamienieni` : label); continue; } const odds = eventOdds(event); if (!odds) { result.unmatched.push(`${label} — brak kursów u bukmacherów`); continue; } result.matched++; const changed = ["odds_home", "odds_draw", "odds_away"].some(k => Math.abs(Number(m[k]) - (odds as any)[k]) >= 0.005); if (!changed && m.odds_api_id === event.id && m.odds_synced_at) continue; const patch: any = { odds_home: odds.odds_home, odds_draw: odds.odds_draw, odds_away: odds.odds_away, odds_api_id: event.id, odds_synced_at: now.toISOString() }; if (!m.odds_synced_at) Object.assign(patch, { odds_open_home: odds.odds_home, odds_open_draw: odds.odds_draw, odds_open_away: odds.odds_away }); await store.update(m.id, patch); if (changed) result.updated++; } } await store.log({ trigger, updated: result.updated, matched: result.matched, unmatched: result.unmatched, credits_remaining: result.credits_remaining, error: result.errors.join("; ") || null, }); return result; } // ── Obsługa wywołania ───────────────────────────────────────────────────────── const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type, x-cron-secret", }; const reply = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...CORS, "Content-Type": "application/json" } }); if (typeof Deno !== "undefined") Deno.serve(async req => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS }); try { return await handle(req); } catch (e) { // Każdy błąd wraca z nagłówkami CORS i treścią — aplikacja pokaże przyczynę zamiast „brak połączenia” console.error("sync-odds:", e); return reply({ ok: false, message: `Błąd funkcji: ${(e as Error)?.message || e}` }, 500); } }); async function handle(req: Request): Promise { const db = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false } }); const apiKey = Deno.env.get("ODDS_API_KEY"); if (!apiKey) return reply({ ok: false, message: "Brak sekretu ODDS_API_KEY w ustawieniach funkcji" }, 500); // Kto wywołuje: harmonogram (sekret z bazy) albo zalogowany admin let trigger: string | null = null; const cron = req.headers.get("x-cron-secret"); if (cron) { const { data } = await db.from("app_secrets").select("value").eq("name", "odds_cron").single(); if (data?.value && data.value === cron) trigger = "cron"; } if (!trigger) { const token = (req.headers.get("Authorization") || "").replace(/^Bearer\s+/i, ""); const { data } = await db.auth.getUser(token); if (data?.user) { const { data: p } = await db.from("profiles").select("is_admin").eq("id", data.user.id).single(); if (p?.is_admin) trigger = "admin"; } } if (!trigger) return reply({ ok: false, message: "Brak uprawnień" }, 401); if (trigger === "admin") { const { data: last } = await db.from("odds_sync_runs").select("ran_at").order("ran_at", { ascending: false }).limit(1).maybeSingle(); const mins = last ? (Date.now() - Date.parse(last.ran_at)) / 60000 : Infinity; if (mins < ADMIN_MIN_GAP_MIN) { return reply({ ok: false, message: `Kursy były aktualizowane ${Math.floor(mins)} min temu. Kolejna aktualizacja za ${Math.ceil(ADMIN_MIN_GAP_MIN - mins)} min.` }); } } const store = { leagues: async () => (await db.from("leagues").select("id, name")).data || [], upcoming: async (from: string, to: string) => (await db.from("matches") .select("id, league_id, home, away, match_date, match_time, status, odds_home, odds_draw, odds_away, odds_manual, odds_api_id, odds_synced_at") .eq("status", "upcoming").gte("match_date", from).lte("match_date", to)).data || [], update: async (id: string, patch: Record) => { await db.from("matches").update(patch).eq("id", id); }, log: async (row: Record) => { await db.from("odds_sync_runs").insert(row); }, }; const result = await syncOdds({ store, apiKey, trigger }); return reply({ ok: true, ...result }); }
